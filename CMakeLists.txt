cmake_minimum_required(VERSION 3.16)

project(BrawlhallaRecomp C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_program(CCACHE ccache)
if(CCACHE)
    set(CMAKE_C_COMPILER_LAUNCHER ${CCACHE})
endif()

set(GENERATED ${CMAKE_SOURCE_DIR}/generated)
set(MOSAIC ${CMAKE_SOURCE_DIR}/tools/mosaic)
foreach(f og_table.c og_image.c libc/og_table.c libc/og_image.c)
    if(NOT EXISTS ${GENERATED}/${f})
        message(FATAL_ERROR "generated/${f} is missing: lift the eboot and libc.prx first (see CLAUDE.md)")
    endif()
endforeach()

function(lifted_library name dir)
    file(GLOB sources CONFIGURE_DEPENDS ${dir}/og_lifted_*.c)
    add_library(${name} STATIC ${sources} ${dir}/og_table.c)
    target_include_directories(${name} PUBLIC ${MOSAIC})
    target_compile_options(${name} PRIVATE -O1 -w)
    if(VITA)
        target_include_directories(${name} PUBLIC ${CMAKE_SOURCE_DIR}/externals/simde)
        target_compile_options(${name} PUBLIC -mfpu=neon -fno-math-errno)
    else()
        target_compile_options(${name} PUBLIC -mavx2)
    endif()
endfunction()
lifted_library(brawlhalla_lifted ${GENERATED})
lifted_library(libc_lifted ${GENERATED}/libc)

file(GLOB ATRAC9_SOURCES externals/LibAtrac9/src/*.c)
add_library(atrac9 STATIC ${ATRAC9_SOURCES})
target_include_directories(atrac9 PUBLIC externals/LibAtrac9/src)
target_compile_options(atrac9 PRIVATE -O2 -w)

file(GLOB RUNTIME_SOURCES CONFIGURE_DEPENDS src/runtime/*.c src/runtime/hle/*.c)
add_executable(brawlhalla ${RUNTIME_SOURCES} ${GENERATED}/og_image.c ${GENERATED}/libc/og_image.c)
target_include_directories(brawlhalla PRIVATE src/runtime externals/stb)
target_compile_definitions(brawlhalla PRIVATE BH_ASSETS_DIR="${CMAKE_SOURCE_DIR}/assets")
target_compile_options(brawlhalla PRIVATE -Wall -Wno-unused-function -g)
option(BH_ASAN "AddressSanitizer on the runtime" OFF)
if(BH_ASAN)
    target_compile_options(brawlhalla PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(brawlhalla PRIVATE -fsanitize=address)
endif()
set_source_files_properties(${GENERATED}/og_image.c PROPERTIES OBJECT_DEPENDS ${GENERATED}/og_image.bin)
set_source_files_properties(${GENERATED}/libc/og_image.c PROPERTIES OBJECT_DEPENDS ${GENERATED}/libc/og_image.bin)

if(VITA)
    include("$ENV{VITASDK}/share/vita.cmake" REQUIRED)
    target_include_directories(brawlhalla PRIVATE $ENV{VITASDK}/arm-vita-eabi/include/SDL2)
    target_link_libraries(brawlhalla brawlhalla_lifted libc_lifted atrac9 SDL2 vita2d SceDisplay_stub SceCtrl_stub SceAudio_stub SceAudioIn_stub
        SceSysmodule_stub SceGxm_stub SceCommonDialog_stub SceAppMgr_stub SceTouch_stub SceHid_stub SceMotion_stub
        SceIme_stub SceAppUtil_stub SceKernelDmacMgr_stub SceIofilemgr_stub SceProcessmgr_stub SceRtc_stub
        ScePower_stub pthread m)
    vita_create_self(eboot.bin brawlhalla UNSAFE)
    vita_create_vpk(Brawlhalla.vpk "CUSA05330" eboot.bin VERSION "10.50" NAME "Brawlhalla")
else()
    find_package(SDL2 REQUIRED)
    find_package(OpenGL REQUIRED)
    target_link_libraries(brawlhalla brawlhalla_lifted libc_lifted atrac9 SDL2::SDL2 OpenGL::GL m)
endif()
